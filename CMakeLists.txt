cmake_minimum_required(VERSION 3.16)
project(ScratchIRCd VERSION 0.18.0 DESCRIPTION "Single-server IRC daemon written from scratch in C" LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_path(ARGON2_INCLUDE_DIR argon2.h)
find_library(ARGON2_LIBRARY NAMES argon2)
if(NOT ARGON2_INCLUDE_DIR OR NOT ARGON2_LIBRARY) message(FATAL_ERROR "libargon2 development files are required") endif()
find_path(MAXMINDDB_INCLUDE_DIR maxminddb.h)
find_library(MAXMINDDB_LIBRARY NAMES maxminddb)
if(NOT MAXMINDDB_INCLUDE_DIR OR NOT MAXMINDDB_LIBRARY) message(FATAL_ERROR "libmaxminddb development files are required") endif()
enable_testing()
add_executable(scratchircd src/main.c src/server.c src/client.c src/channel.c src/channel_policy.c src/hash.c src/irc.c src/ircv3.c src/runtime_config.c src/dns.c src/dnsbl.c src/geoip.c src/history_db.c src/mail.c src/modes.c src/oper.c src/operator_db.c src/ban_db.c src/nickserv.c src/nickserv_db.c src/chanserv.c src/chanserv_db.c src/visibility.c src/commands/admin.c src/commands/authenticate.c src/commands/away.c src/commands/cap.c src/commands/chathistory.c src/commands/chanserv.c src/commands/chanserv_admin.c src/commands/common.c src/commands/dispatch.c src/commands/identify.c src/commands/invite.c src/commands/ison.c src/commands/joinpart.c src/commands/kick.c src/commands/kill.c src/commands/kline.c src/commands/list.c src/commands/lusers.c src/commands/mode.c src/commands/motd.c src/commands/names.c src/commands/nick.c src/commands/nickserv.c src/commands/nickserv_admin.c src/commands/notice.c src/commands/oper.c src/commands/operator_admin.c src/commands/pass.c src/commands/ping.c src/commands/privmsg.c src/commands/quit.c src/commands/rehash.c src/commands/restart.c src/commands/rules.c src/commands/sajoin.c src/commands/samode.c src/commands/sapart.c src/commands/sethost.c src/commands/setident.c src/commands/setname.c src/commands/topic.c src/commands/user.c src/commands/userhost.c src/commands/userip.c src/commands/wallops.c src/commands/webirc.c src/commands/who.c src/commands/whois.c src/commands/zline.c)
target_include_directories(scratchircd PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include ${ARGON2_INCLUDE_DIR} ${MAXMINDDB_INCLUDE_DIR})
target_compile_definitions(scratchircd PRIVATE _POSIX_C_SOURCE=200809L)
target_link_libraries(scratchircd PRIVATE Threads::Threads ${ARGON2_LIBRARY} ${MAXMINDDB_LIBRARY} SQLite::SQLite3 OpenSSL::SSL OpenSSL::Crypto)
add_executable(scratchircd-mkpasswd tools/mkpasswd.c);target_include_directories(scratchircd-mkpasswd PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include ${ARGON2_INCLUDE_DIR});target_link_libraries(scratchircd-mkpasswd PRIVATE ${ARGON2_LIBRARY})
function(simple_test target name)
 add_test(NAME ${name} COMMAND ${target})
endfunction()
add_executable(test_client_identity tests/test_client_identity.c src/client.c);target_include_directories(test_client_identity PRIVATE include ${MAXMINDDB_INCLUDE_DIR});target_link_libraries(test_client_identity PRIVATE OpenSSL::SSL OpenSSL::Crypto);simple_test(test_client_identity client_identity)
add_executable(test_geoip tests/test_geoip.c src/geoip.c);target_include_directories(test_geoip PRIVATE include ${MAXMINDDB_INCLUDE_DIR});target_link_libraries(test_geoip PRIVATE ${MAXMINDDB_LIBRARY});simple_test(test_geoip geoip)
add_executable(test_dnsbl tests/test_dnsbl.c src/dnsbl.c);target_include_directories(test_dnsbl PRIVATE include);target_compile_definitions(test_dnsbl PRIVATE _POSIX_C_SOURCE=200809L);target_link_libraries(test_dnsbl PRIVATE Threads::Threads);simple_test(test_dnsbl dnsbl)
add_executable(test_runtime_config tests/test_runtime_config.c src/runtime_config.c);target_include_directories(test_runtime_config PRIVATE include);target_compile_definitions(test_runtime_config PRIVATE _POSIX_C_SOURCE=200809L);simple_test(test_runtime_config runtime_config)
add_executable(test_modes tests/test_modes.c src/modes.c);target_include_directories(test_modes PRIVATE include);simple_test(test_modes modes)
add_executable(test_channel_policy tests/test_channel_policy.c src/client.c src/channel.c src/channel_policy.c);target_include_directories(test_channel_policy PRIVATE include ${MAXMINDDB_INCLUDE_DIR});target_link_libraries(test_channel_policy PRIVATE OpenSSL::SSL OpenSSL::Crypto);simple_test(test_channel_policy channel_policy)
add_executable(test_visibility tests/test_visibility.c src/client.c src/channel.c src/channel_policy.c src/modes.c src/visibility.c);target_include_directories(test_visibility PRIVATE include ${MAXMINDDB_INCLUDE_DIR});target_link_libraries(test_visibility PRIVATE OpenSSL::SSL OpenSSL::Crypto);simple_test(test_visibility visibility)
add_executable(test_oper tests/test_oper.c src/oper.c);target_include_directories(test_oper PRIVATE include);simple_test(test_oper oper_permissions)
add_executable(test_operator_db tests/test_operator_db.c src/operator_db.c);target_include_directories(test_operator_db PRIVATE include);target_compile_definitions(test_operator_db PRIVATE _POSIX_C_SOURCE=200809L);target_link_libraries(test_operator_db PRIVATE SQLite::SQLite3);simple_test(test_operator_db operator_database)
add_executable(test_ban_db tests/test_ban_db.c src/ban_db.c);target_include_directories(test_ban_db PRIVATE include);target_compile_definitions(test_ban_db PRIVATE _POSIX_C_SOURCE=200809L);target_link_libraries(test_ban_db PRIVATE SQLite::SQLite3);simple_test(test_ban_db ban_database)
add_executable(test_history_db tests/test_history_db.c src/history_db.c);target_include_directories(test_history_db PRIVATE include);target_compile_definitions(test_history_db PRIVATE _POSIX_C_SOURCE=200809L);target_link_libraries(test_history_db PRIVATE SQLite::SQLite3);simple_test(test_history_db history_database)
add_executable(test_nickserv_db tests/test_nickserv_db.c src/nickserv_db.c);target_include_directories(test_nickserv_db PRIVATE include);target_compile_definitions(test_nickserv_db PRIVATE _POSIX_C_SOURCE=200809L);target_link_libraries(test_nickserv_db PRIVATE SQLite::SQLite3 OpenSSL::Crypto);simple_test(test_nickserv_db nickserv_database)
add_executable(test_chanserv_db tests/test_chanserv_db.c src/chanserv_db.c);target_include_directories(test_chanserv_db PRIVATE include);target_compile_definitions(test_chanserv_db PRIVATE _POSIX_C_SOURCE=200809L);target_link_libraries(test_chanserv_db PRIVATE SQLite::SQLite3);simple_test(test_chanserv_db chanserv_database)
foreach(t protocol operator_actions operator_override tls webirc nickserv sasl ircv3_cap history chanserv)
 if(t STREQUAL "protocol" OR t STREQUAL "operator_actions" OR t STREQUAL "operator_override" OR t STREQUAL "nickserv")
  add_test(NAME ${t}_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_${t}.py $<TARGET_FILE:scratchircd> $<TARGET_FILE:scratchircd-mkpasswd>)
 else()
  add_test(NAME ${t}_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_${t}.py $<TARGET_FILE:scratchircd>)
 endif()
 set_tests_properties(${t}_integration PROPERTIES TIMEOUT 120)
endforeach()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 foreach(target scratchircd scratchircd-mkpasswd test_client_identity test_geoip test_dnsbl test_runtime_config test_modes test_channel_policy test_visibility test_oper test_operator_db test_ban_db test_history_db test_nickserv_db test_chanserv_db) target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic) endforeach()
endif()
install(TARGETS scratchircd scratchircd-mkpasswd RUNTIME DESTINATION bin)
